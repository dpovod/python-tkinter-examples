Текстове поле та кнопка

Ось приклад коду для створення вікна з кнопкою та текстовим полем за допомогою Tkinter:

{% highlight python %} import tkinter import tkinter.messagebox # Створення вікна програми window = tkinter.Tk() # Створення змінної для отримання тексту з поля вводу str_var = tkinter.StringVar() # Функція обробки натискання кнопки def button_click(): tkinter.messagebox.showinfo("Увага!", "Сьогодні - " + str_var.get()) # Створення текстового напису label = tkinter.Label(text="Введіть день тижня:") label.pack() # Створення текстового поля для введення дня тижня edit = tkinter.Entry(window, textvariable=str_var) edit.pack() # Створення кнопки для підтвердження button = tkinter.Button(window, text="Гаразд", command=button_click) button.pack() # Запуск головного циклу програми window.mainloop() {% endhighlight %}
